36template <class T> void selectionSort(T arr[], int size) {
37 for (int i = 0; i < size - 1; i++) {
38 int minIndex = i;
39
40 for (int j = i + 1; j < size; j++) {
41 if (arr[j] < arr[minIndex]) {
42 minIndex = j;
43 }
44 }
45
46 if (minIndex != i) {
47 swap(arr[i], arr[minIndex]);
48 }
49 }
50}

63template <class T> void insertionSort(T arr[], int size) {
64 for (int i = 1; i < size; i++) {
65 T current = arr[i];
66 int j = i - 1;
67
68 while (j >= 0 && arr[j] > current) {
69 arr[j + 1] = arr[j];
70 j--;
71 }
72
73 arr[j + 1] = current;
74 }
75}

89template <class T> void heapify(T arr[], int size, int root) {
90 int largest = root;
91 int left = 2 * root + 1;
92 int right = 2 * root + 2;
93
94 if (left < size && arr[left] > arr[largest]) {
95 largest = left;
96 }
97
98 if (right < size && arr[right] > arr[largest]) {
99 largest = right;
100 }
101
102 if (largest != root) {
103 swap(arr[root], arr[largest]);
104 heapify(arr, size, largest);
105 }
106}

118template <class T> void heapSort(T arr[], int size) {
119 for (int i = size / 2 - 1; i >= 0; i--) {
120 heapify(arr, size, i);
121 }
122
123 for (int i = size - 1; i > 0; i--) {
124 swap(arr[0], arr[i]);
125 heapify(arr, i, 0);
