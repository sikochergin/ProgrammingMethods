46template <typename SearchFunc>
47double measureSearch(SearchFunc search, const vector<string>& keys, int measureRepeats = 5)
48{
49 double best = 1e18;
50
51 for (int r = 0; r < measureRepeats; ++r) {
52 long long sink = 0;
53
54 auto start = high_resolution_clock::now();
55 for (const string& key : keys) {
56 sink += search(key).size();
57 }
58 auto end = high_resolution_clock::now();
59
60 volatile long long keep = sink;
61 (void)keep;
62
63 double ns = duration<double, nano>(end - start).count() / keys.size();
64 if (ns < best) {
65 best = ns;
66 }
67 }
68
69 return best;
70}

84vector<string> makeQueryKeys(const vector<Soldier>& soldiers, int count, mt19937& rng)
85{
86 vector<string> keys;
87 uniform_int_distribution<size_t> pick(0, soldiers.size() - 1);
88
89 for (int i = 0; i < count; ++i) {
90 if (i % 10 == 0) {
91 keys.push_back("НЕСУЩЕСТВУЮЩЕЕ ФИО");
92 }
93 else {
94 keys.push_back(soldiers[pick(rng)].getFullName());
95 }
96 }
97
98 return keys;
99}

110void runTimeBenchmark(const vector<int>& sizes)
111{
112 ofstream out("search_time.csv");
113 if (!out.is_open()) {
114 cout << "Ошибка: не удалось открыть search_time.csv" << endl;
115 return;
116 }
117 out << "run,size,algorithm,nanoseconds\n";
118
119 mt19937 rng(42);
120 const int QUERIES = 50; // не менее 10 поисков на каждой выборке
121
122 for (int size : sizes) {
123 for (int run = 0; run < 5; ++run) {
124 string fileName = "mil_csv_folder/soldiers_" + to_string(size) + "_" + to_string(run) + ".csv";
125 vector<Soldier> soldiers = readSoldiersFromCSV(fileName);
126
127 if (soldiers.empty()) {
128 cout << "Пропущен файл: " << fileName << endl;
129 continue;
130 }
131
132 // Размер хэш-таблицы с запасом относительно числа записей —
133 // для быстрого поиска (низкий коэффициент загрузки).
134 HashTable hash(size);
135
136 // Все структуры строим вне замера времени поиска.
137 BinarySearchTree bst;
138 bst.build(soldiers);
139
140 RedBlackTree rbt;
141 rbt.build(soldiers);
142
143 hash.build(soldiers);
144
145 multimap<string, Soldier> mm = buildMultiMap(soldiers);
146
147 vector<string> keys = makeQueryKeys(soldiers, QUERIES, rng);
148
149 double tLinear = measureSearch(
150 [&](const string& k) { return linearSearchByFullName(soldiers, k); }, keys);
151 double tBst = measureSearch(
152 [&](const string& k) { return bst.search(k); }, keys);
153 double tRbt = measureSearch(
154 [&](const string& k) { return rbt.search(k); }, keys);
155 double tHash = measureSearch(
156 [&](const string& k) { return hash.search(k); }, keys);
157 double tMm = measureSearch(
158 [&](const string& k) { return searchInMultiMap(mm, k); }, keys);
159
160 out << run << "," << size << ",linear_search," << tLinear << "\n";
161 out << run << "," << size << ",binary_search_tree," << tBst << "\n";
162 out << run << "," << size << ",red_black_tree," << tRbt << "\n";
163 out << run << "," << size << ",hash_table," << tHash << "\n";
164 out << run << "," << size << ",multimap," << tMm << "\n";
165
166 cout << "Время: size=" << size << " run=" << run << " готово" << endl;
167 }
168 }
169
170 out.close();
171 cout << "Время поиска записано в search_time.csv" << endl;
172}

186void runCollisionStudy(const vector<int>& sizes)
187{
188 ofstream out("collisions.csv");
189 if (!out.is_open()) {
190 cout << "Ошибка: не удалось открыть collisions.csv" << endl;

192 }
193 out << "run,size,collisions\n";
194
195 const int FIXED_TABLE_SIZE = 20011; // простое число, с запасом над числом уникальных ФИО
196
197 for (int size : sizes) {
198 for (int run = 0; run < 5; ++run) {
199 string fileName = "mil_csv_folder/soldiers_" + to_string(size) + "_" + to_string(run) + ".csv";
200 vector<Soldier> soldiers = readSoldiersFromCSV(fileName);
201
202 if (soldiers.empty()) {
203 continue;
204 }
205
206 HashTable hash(FIXED_TABLE_SIZE);
207 hash.build(soldiers);
208
209 out << run << "," << size << "," << hash.getCollisionCount() << "\n";
210 }
211 cout << "Коллизии: size=" << size << " готово" << endl;
212 }
213
214 out.close();
215 cout << "Коллизии записаны в collisions.csv" << endl;
216}
