125double calculateStddev(const std::vector<double>& values, double mean) {
126 if (values.size() < 2){
127 return 0.0;
128 }
129
130 double sumSquares = 0.0;
131
132 for (double value : values) {
133 double diff = value - mean;
134 sumSquares += diff * diff;
135 }
136
137 return std::sqrt(sumSquares / (values.size() - 1));

251double chiSquareCriticalValue(int degreesOfFreedom, double alpha) {
252 if (degreesOfFreedom <= 0) {
253 return 0.0;
254 }
255
256 double k = static_cast<double>(degreesOfFreedom);
257 double z = inverseNormalCDF(1.0 - alpha);
258
259 double value = 1.0 - 2.0 / (9.0 * k) + z * std::sqrt(2.0 / (9.0 * k));
260
261 return k * value * value * value;
262}

438TestResult overlappingPermutationsTest(const std::vector<double>& values) {
439 int permutationFrequency[120];
440
441 for (int i = 0; i < 120; i++) {
442 permutationFrequency[i] = 0;
443 }
444
445 int factorials[5] = {1, 1, 2, 6, 24};
446
447 int totalWindows = 0;
448
449 for (int start = 0; start + 5 <= static_cast<int>(values.size()); start++) {
450 std::pair<double, int> window[5];
451
452 for (int position = 0; position < 5; position++) {
453 window[position] = std::make_pair(values[start + position], position);
454 }
455
456 std::sort(window, window + 5);
457
458 int permutation[5];
459
460 for (int sortedPosition = 0; sortedPosition < 5; sortedPosition++) {
461 permutation[sortedPosition] = window[sortedPosition].second;
462 }
463
464 int permutationIndex = 0;
465
466 for (int i = 0; i < 5; i++) {
467 int smallerOnRight = 0;
468
469 for (int j = i + 1; j < 5; j++) {
470 if (permutation[j] < permutation[i]) {
471 smallerOnRight++;
472 }
473 }
474
475 permutationIndex += smallerOnRight * factorials[4 - i];
476 }
477
478 permutationFrequency[permutationIndex]++;
479 totalWindows++;
480 }
481
482 double expected = totalWindows / 120.0;
483 double chiSquare = 0.0;
484
485 for (int i = 0; i < 120; i++) {
486 double diff = permutationFrequency[i] - expected;
487 chiSquare += diff * diff / expected;
488 }
489
490 double criticalValue = chiSquareCriticalValue(119, 0.05);
491 bool passed = chiSquare <= criticalValue;
492
493 return {"OverlappingPermutations", chiSquare, passed};

525ChiSquareResult chiSquareUniformTest(const std::vector<double>& values) {
526 int binCount = static_cast<int>(1.0 + std::log2(values.size()));
527 if (binCount < 5){
528 binCount = 5;
529 }
530
531 std::vector<int> bins(binCount, 0);
532
533 for (double value : values) {
534 int index = static_cast<int>(value * binCount);
535
536 if (index == binCount) {
537 index = binCount - 1;
538 }
539
540 bins[index]++;
541 }
542
543 double expected = values.size() / static_cast<double>(binCount);
544 double chiSquare = 0.0;
545
546 for (int count : bins) {
547 double diff = count - expected;
548 chiSquare += diff * diff / expected;
549 }
550
551 double criticalValue = chiSquareCriticalValue(binCount - 1, 0.05);
552
553 bool passed = chiSquare <= criticalValue;
554
555 return {chiSquare, criticalValue, passed};
